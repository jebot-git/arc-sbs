cmake_minimum_required(VERSION 3.21)
project(arc-sbs-fltk LANGUAGES CXX)
option(ARC_SBS_ENABLE_X11 "Allow explicitly selected X11 frontend" OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig zlib)
add_executable(arc-sbs-fltk main.cpp)
target_compile_features(arc-sbs-fltk PRIVATE cxx_std_17)
target_compile_definitions(arc-sbs-fltk PRIVATE ARC_SBS_ENABLE_X11=$<BOOL:${ARC_SBS_ENABLE_X11}>)
set_target_properties(arc-sbs-fltk PROPERTIES INSTALL_RPATH "$<TARGET_FILE_DIR:fltk::fltk-shared>")
target_compile_options(arc-sbs-fltk PRIVATE -Wall -Wextra)
target_link_libraries(arc-sbs-fltk PRIVATE fltk::fltk-shared fltk::gl-shared fltk::images-shared PkgConfig::FONTCONFIG)
include(GNUInstallDirs)
install(TARGETS arc-sbs-fltk RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

include(desktop-menu.cmake)
de_xr_desktop_menu(arc-sbs-fltk)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  if(ARC_SBS_ENABLE_X11)
    add_executable(arc-sbs-ui-test tests/frontend.cpp)
    target_compile_features(arc-sbs-ui-test PRIVATE cxx_std_17)
    target_compile_definitions(arc-sbs-ui-test PRIVATE ARC_SBS_ENABLE_X11=1)
    target_link_libraries(arc-sbs-ui-test PRIVATE fltk::fltk-shared fltk::gl-shared fltk::images-shared PkgConfig::FONTCONFIG)
    de_xr_desktop_menu(arc-sbs-ui-test)
    add_test(NAME native-frontend COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/frontend.py $<TARGET_FILE:arc-sbs-ui-test>)
  else()
    add_test(NAME wayland-default COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/backend.py $<TARGET_FILE:arc-sbs-fltk>)
  endif()
endif()
